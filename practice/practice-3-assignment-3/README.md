#Ответы на вопросы
# Создание датафрейма
id <- 1:6                           # ID студентов
name <- c("Anna", "Boris", "Clara", "Dmitry", "Elena", "Fedor") # имена
age <- c(19, 22, 20, 23, 21, 22)   # возраст
rooms <- c(5, 7, 8, 6, 9, 4)        # количество комнат
score <- c(75, 88, 92, 70, 95, 60)  # баллы
df <- data.frame(id, name, age, rooms, score)  # создаём датафрейм
df                                  # выводим датафрейм

# Удаление столбцов
df_minus1 <- df[, -1]               # удалить первый столбец
df_minus2 <- df[, -2]               # удалить второй столбец

# Получение столбца и сохранение в вектор
v1 <- df$rooms                       # через $
v2 <- df[, "rooms"]                  # через имя столбца
v3 <- df[["rooms"]]                  # через двойные скобки

# Выбор данных по условию
subset(df, rooms > 7)                # комнаты > 7 (subset)
df[df$rooms > 7, ]                   # комнаты > 7 (логическая индексация)

# Сортировка датафрейма
df[order(df$score), ]                # сортировка по score по возрастанию
df[order(-df$score), ]               # сортировка по score по убыванию
